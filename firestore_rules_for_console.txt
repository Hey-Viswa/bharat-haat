rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    // Helper functions
    function isAuthenticated() {
      return request.auth != null;
    }

    function isOwner(uid) {
      return request.auth.uid == uid;
    }

    // Users collection - each user can only access their own profile
    match /users/{userId} {
      // Allow full read/write access to own profile - simplified for better compatibility
      allow read, write: if isAuthenticated() && isOwner(userId);

      // Allow reading seller profiles for public display (limited fields)
      allow read: if isAuthenticated() &&
                     resource.data.role == "seller" &&
                     request.query.limit <= 50;

      // Allow creating user profile - more permissive for initial setup
      allow create: if isAuthenticated() &&
                       isOwner(userId);

      // Allow updating user profile - simplified ownership check
      allow update: if isAuthenticated() && isOwner(userId);
    }

    // Products collection
    match /products/{productId} {
      // Public read access for browsing products (with limits)
      allow read: if isAuthenticated() &&
                     resource.data.is_active == true &&
                     request.query.limit <= 100;

      // Sellers can only manage their own products
      allow create: if isAuthenticated() &&
                       request.resource.data.seller_id == request.auth.uid;

      allow update, delete: if isAuthenticated() &&
                               resource.data.seller_id == request.auth.uid;

      // Allow sellers to read their own products (including inactive ones)
      allow read: if isAuthenticated() &&
                     resource.data.seller_id == request.auth.uid;
    }

    // Categories collection (read-only for clients)
    match /categories/{categoryId} {
      allow read: if isAuthenticated();
      allow write: if false; // Admin-only through backend
    }

    // Reviews collection
    match /reviews/{reviewId} {
      // Public read access to reviews
      allow read: if isAuthenticated();

      // Users can create reviews
      allow create: if isAuthenticated() &&
                       request.resource.data.user_id == request.auth.uid;

      // Users can update/delete their own reviews
      allow update, delete: if isAuthenticated() &&
                               resource.data.user_id == request.auth.uid;
    }

    // Wishlist collection
    match /wishlists/{userId} {
      allow read, write: if isAuthenticated() && isOwner(userId);
    }

    // Cart collection
    match /carts/{userId} {
      allow read, write: if isAuthenticated() && isOwner(userId);
    }
  }
}

// Firebase Storage Security Rules
service firebase.storage {
  match /b/{bucket}/o {

    // Product images - sellers can only upload to their own folder
    match /products/{sellerId}/{allPaths=**} {
      allow read: if true; // Public read access for product images
      allow write: if request.auth != null &&
                      request.auth.uid == sellerId &&
                      request.resource.size < 5 * 1024 * 1024 && // 5MB limit
                      request.resource.contentType.matches('image/.*');
    }

    // User profile pictures
    match /users/{userId}/profile/{fileName} {
      allow read: if true; // Public read access for profile pictures
      allow write: if request.auth != null &&
                      request.auth.uid == userId &&
                      request.resource.size < 2 * 1024 * 1024 && // 2MB limit
                      request.resource.contentType.matches('image/.*');
    }
  }
}
